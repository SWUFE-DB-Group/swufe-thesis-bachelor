\section{数学公式编写}

规范要求电子稿公式需编号清晰。学术论文中，公式通常采用按章独立连续编号的形式，即形如“(2-1)”。

例如，著名的注意力机制（Attention Mechanism）\cite{Transformer}计算公式如下：
\begin{equation}
\label{eq:attention}
\mathrm{Attention}(Q, K, V) = \mathrm{softmax}\left(\frac{QK^T}{\sqrt{d_k}}\right)V
\end{equation}
其中 $Q, K, V$ 分别表示查询向量（Query）、键向量（Key）和值向量（Value），$d_k$ 为键向量的特征维度。

再如多元高斯分布的概率密度函数公式：
\begin{equation}
\label{eq:multivariate-gaussian}
p(\mathbf{x}; \boldsymbol{\mu}, \boldsymbol{\Sigma}) = \frac{1}{(2\pi)^{d/2}|\boldsymbol{\Sigma}|^{1/2}} \exp\left(-\frac{1}{2}(\mathbf{x}-\boldsymbol{\mu})^T \boldsymbol{\Sigma}^{-1} (\mathbf{x}-\boldsymbol{\mu})\right)
\end{equation}
公式~\eqref{eq:attention} 与公式~\eqref{eq:multivariate-gaussian} 均符合国家标准数学符号规范。

\subsection{算法排版}
模板提供了 \texttt{algorithm2e} 宏包支持，可以撰写带行号、规范规整的伪代码：

\begin{algorithm}[H]
\caption{基于动量的随机梯度下降算法 (SGD with Momentum)}
\label{alg:momentum-sgd}
\KwIn{学习率 $\eta$，动量衰减系数 $\gamma \in [0, 1)$，初始模型参数 $\theta_0$}
\KwOut{优化后的模型参数 $\theta^*$}
初始化速度向量 $v_0 \leftarrow 0$\;
\For{$t = 1, 2, \ldots, T$}{
    从训练集中随机采样一个批次样本 $\mathcal{B}_t$\;
    计算小批量样本上的损失函数梯度 $g_t \leftarrow \nabla_\theta \mathcal{L}(\theta_{t-1}; \mathcal{B}_t)$\;
    更新动量速度：$v_t \leftarrow \gamma v_{t-1} + \eta g_t$\;
    更新模型参数：$\theta_t \leftarrow \theta_{t-1} - v_t$\;
    \If{$\|\theta_t - \theta_{t-1}\| < \epsilon$}{
        提前停止迭代\;
        \textbf{break}\;
    }
}
\Return{$\theta_T$}\;
\end{algorithm}

\begin{codeblock}[language=Python]
import torch
import torch.nn as nn

class ResidualBlock(nn.Module):
    """ResNet 残差连接模块示例"""
    def __init__(self, channels: int):
        super().__init__()
        self.conv1 = nn.Conv2d(channels, channels, kernel_size=3, padding=1)
        self.bn1 = nn.BatchNorm2d(channels)
        self.relu = nn.ReLU(inplace=True)
        self.conv2 = nn.Conv2d(channels, channels, kernel_size=3, padding=1)
        self.bn2 = nn.BatchNorm2d(channels)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        residual = x
        out = self.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out += residual  # 残差跳跃连接
        return self.relu(out)
\end{codeblock}

\section{数学推导与定理环境}

\subsection{多行公式、分段函数与矩阵}

多行推导可使用 \texttt{align} 环境，在等号处用符号 \texttt{\&} 对齐。
例如，二次损失函数的梯度可写为
\begin{align}
\mathcal{L}(\boldsymbol{w})
  &= \frac{1}{2n}\lVert X\boldsymbol{w}-\boldsymbol{y}\rVert_2^2,
  \label{eq:quadratic-loss} \\
\nabla\mathcal{L}(\boldsymbol{w})
  &= \frac{1}{n}X^{\mathsf{T}}(X\boldsymbol{w}-\boldsymbol{y}).
  \label{eq:quadratic-gradient}
\end{align}
式~\eqref{eq:quadratic-loss} 与式~\eqref{eq:quadratic-gradient} 分别展示逐行编号和交叉引用。

分段函数使用 \texttt{cases}，例如二分类决策规则
\begin{equation}
\widehat{y}(s)=
\begin{cases}
1, & s\geq\tau, \\
0, & s<\tau,
\end{cases}
\label{eq:threshold-rule}
\end{equation}
其中 $s\in[0,1]$ 为模型输出的风险分数，$\tau$ 为判定阈值。
矩阵可使用 \texttt{bmatrix}，例如一个无向图的邻接矩阵
\begin{equation}
A=\begin{bmatrix}
0 & 1 & 1 \\
1 & 0 & 0 \\
1 & 0 & 0
\end{bmatrix}.
\label{eq:adjacency-matrix}
\end{equation}

\subsection{定义、引理、定理与证明}

模板采用 SJTUTeX 对 \texttt{amsthm} 的支持。
定义、引理、定理、推论和例分别按章独立编号，证明与注不编号；
可选参数用于填写名称，标签与普通公式一样可以交叉引用。

\begin{definition}[欧氏范数]
\label{def:euclidean-norm}
对向量 $\boldsymbol{x}=(x_1,\ldots,x_d)^{\mathsf{T}}\in\mathbb{R}^d$，
其欧氏范数定义为
\[
\lVert\boldsymbol{x}\rVert_2
  =\left(\sum_{j=1}^{d}x_j^2\right)^{1/2}.
\]
\end{definition}

\begin{lemma}[平方和恒等式]
\label{lem:pairwise-distance}
设 $\boldsymbol{x}_1,\ldots,\boldsymbol{x}_n\in\mathbb{R}^d$，其中 $n\geq1$，则
\begin{equation}
\sum_{i=1}^{n}\sum_{j=1}^{n}
  \lVert\boldsymbol{x}_i-\boldsymbol{x}_j\rVert_2^2
  =2n\sum_{i=1}^{n}\lVert\boldsymbol{x}_i\rVert_2^2
   -2\left\lVert\sum_{i=1}^{n}\boldsymbol{x}_i\right\rVert_2^2.
\label{eq:pairwise-distance}
\end{equation}
\end{lemma}

\begin{proof}
由定义~\ref{def:euclidean-norm}，展开每一项可得
\[
\lVert\boldsymbol{x}_i-\boldsymbol{x}_j\rVert_2^2
=\lVert\boldsymbol{x}_i\rVert_2^2
+\lVert\boldsymbol{x}_j\rVert_2^2
-2\boldsymbol{x}_i^{\mathsf{T}}\boldsymbol{x}_j.
\]
对 $i,j$ 求和，并注意交叉项之和等于向量和的范数平方，即得结论。
\end{proof}

\begin{theorem}[均值向量的范数界]
\label{thm:mean-norm-bound}
对任意 $\boldsymbol{x}_1,\ldots,\boldsymbol{x}_n\in\mathbb{R}^d$，其中 $n\geq1$，有
\begin{equation}
\left\lVert\frac{1}{n}\sum_{i=1}^{n}\boldsymbol{x}_i\right\rVert_2^2
\leq\frac{1}{n}\sum_{i=1}^{n}\lVert\boldsymbol{x}_i\rVert_2^2.
\label{eq:mean-norm-bound}
\end{equation}
等号成立当且仅当所有向量相同。
\end{theorem}

\begin{proof}
由引理~\ref{lem:pairwise-distance}，下式右端非负；等号成立恰好要求每一对向量相同：
\begin{align*}
\frac{1}{n}\sum_{i=1}^{n}\lVert\boldsymbol{x}_i\rVert_2^2
-\left\lVert\frac{1}{n}\sum_{i=1}^{n}\boldsymbol{x}_i\right\rVert_2^2
&=\frac{1}{2n^2}\sum_{i=1}^{n}\sum_{j=1}^{n}
  \lVert\boldsymbol{x}_i-\boldsymbol{x}_j\rVert_2^2 \\
&\geq0.\qedhere
\end{align*}
\end{proof}

\begin{corollary}[两个向量的情形]
\label{cor:two-vector-mean}
在定理~\ref{thm:mean-norm-bound} 中取 $n=2$，可得
\[
\left\lVert\frac{\boldsymbol{x}+\boldsymbol{y}}{2}\right\rVert_2^2
\leq\frac{\lVert\boldsymbol{x}\rVert_2^2+\lVert\boldsymbol{y}\rVert_2^2}{2}.
\]
\end{corollary}

\begin{example}[二维向量]
取 $\boldsymbol{x}=(1,0)^{\mathsf{T}}$、$\boldsymbol{y}=(0,1)^{\mathsf{T}}$，
推论~\ref{cor:two-vector-mean} 两侧分别为 $1/2$ 和 $1$。
\end{example}

\begin{remark}
证明以公式结束时，可用 \texttt{\textbackslash qedhere} 对齐证毕符号。
\end{remark}
